// =============================================
// CSV パース
// =============================================
function parseCSV(text) {
  const lines = text.trim().split('\n');
  if (lines.length < 2) return [];
  const headers = lines[0].split(',').map(h => h.replace(/^"|"$/g,'').trim());
  return lines.slice(1).map(line => {
    const cols = []; let cur='', inQ=false;
    for (let c of line) {
      if (c==='"') inQ=!inQ;
      else if (c===',' && !inQ) { cols.push(cur); cur=''; }
      else cur+=c;
    }
    cols.push(cur);
    const row={};
    headers.forEach((h,i) => row[h]=(cols[i]||'').trim());
    return row;
  });
}

// =============================================
// データ分析
// =============================================
function analyze(rows) {
  const total = rows.length;
  const wearCount = {}; PATTERNS.forEach(p=>wearCount[p.name]=0);
  const symCount = {}; SYMPTOMS.forEach(s=>symCount[s]=0);
  const wearSym = {}; PATTERNS.forEach(p=>{wearSym[p.name]={}; SYMPTOMS.forEach(s=>wearSym[p.name][s]=0);});
  const ageCount={}, genderCount={}, shoeCount={};

  rows.forEach(row => {
    const wearKey = Object.keys(row).find(k=>k.includes('摩耗パターン'));
    const symKey  = Object.keys(row).find(k=>k.includes('現在ある症状') && !k.includes('その他'));
    const ageKey  = Object.keys(row).find(k=>k.includes('年齢'));
    const genderKey = Object.keys(row).find(k=>k.includes('性別'));
    const shoeKey = Object.keys(row).find(k=>k.includes('靴の種類'));

    const wear = wearKey ? row[wearKey] : '';
    const syms = symKey  ? row[symKey]  : '';
    const age  = ageKey  ? row[ageKey]  : '';
    const gender = genderKey ? row[genderKey] : '';
    const shoe = shoeKey ? row[shoeKey] : '';

    if (wear && wearCount[wear]!==undefined) wearCount[wear]++;
    SYMPTOMS.forEach(s => {
      if (syms.includes(s)) {
        symCount[s]++;
        if (wear && wearSym[wear]) wearSym[wear][s]++;
      }
    });
    if (age)    ageCount[age]    = (ageCount[age]||0)+1;
    if (gender) genderCount[gender] = (genderCount[gender]||0)+1;
    if (shoe)   shoeCount[shoe]  = (shoeCount[shoe]||0)+1;
  });

  return { total, wearCount, symCount, wearSym, ageCount, genderCount, shoeCount };
}

function generateInsights(sorted, wearCount, symCount, total) {
  if (total===0) return '';
  const top = sorted[0];
  const topPct = Math.round((wearCount[top.name]||0)/total*100);
  const topSym = Object.entries(symCount).sort((a,b)=>b[1]-a[1])[0];
  const femaleCount = 0; // データから取得
  return `
    <div class="insight-card red">
      <div class="insight-title">🏆 最多摩耗パターン</div>
      <div class="insight-text">${top.name}が全体の${topPct}%（${wearCount[top.name]||0}件）を占めます。</div>
    </div>
    <div class="insight-card blue">
      <div class="insight-title">🔴 最多症状</div>
      <div class="insight-text">${topSym[0]}が${topSym[1]}件（${Math.round(topSym[1]/total*100)}%）と最多です。</div>
    </div>
    <div class="insight-card yellow">
      <div class="insight-title">💡 平均症状数</div>
      <div class="insight-text">1人あたり平均<strong>${(Object.values(symCount).reduce((s,v)=>s+v,0)/total).toFixed(1)}種類</strong>の症状があります。</div>
    </div>`;
}

function switchTab(id, btn) {
  document.querySelectorAll('.tab-panel').forEach(p=>p.classList.remove('active'));
  document.querySelectorAll('.tab-btn').forEach(b=>b.classList.remove('active'));
  document.getElementById('tab-'+id).classList.add('active');
  btn.classList.add('active');
}

async function loadData() {
  document.getElementById('main-content').innerHTML = `
    <div class="loading-wrap"><div class="spinner"></div><div class="loading-text">データ取得中...</div></div>`;
  try {
    const res = await fetch(CSV_URL);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const text = await res.text();
    const rows = parseCSV(text);
    render(analyze(rows));
  } catch(e) {
    document.getElementById('main-content').innerHTML = `
      <div class="error-box">
        <div class="error-title">⚠️ データ取得に失敗しました</div>
        <div class="error-text">
          スプレッドシートの公開設定を確認してください。<br><br>
          スプレッドシートアプリ →「⋯」→「共有とエクスポート」→「アクセスを管理」→「リンクを知っている全員・閲覧者」<br><br>
          エラー: ${e.message}
        </div>
      </div>
      <button class="refresh-btn" onclick="loadData()" style="margin-top:8px;">🔄 再試行する</button>`;
  }
}
